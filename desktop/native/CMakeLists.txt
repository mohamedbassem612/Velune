cmake_minimum_required(VERSION 3.21)
project(VeluneAudio VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(velune-audio mpv-host.cpp)
target_include_directories(velune-audio PRIVATE vendor)
target_link_libraries(velune-audio PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(velune-audio PRIVATE /utf-8 /W4)
else()
  target_compile_options(velune-audio PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS velune-audio RUNTIME DESTINATION native)
